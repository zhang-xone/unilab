#   algo.max_iterations=200 : 训练200轮 #
#   algo.num_steps_per_env=24 : 每环境每轮采集24步
#   algo.num_envs=4096   : 4096个并行环境
#   algo.save_interval=50 : 每50轮保存一次checkpoint
#   algo.load_run="-1"   : 加载最新训练结果（继续训练）
#   algo.checkpoint=-1   : 使用最新检查点
#   training.no_play=true : 关闭回放（加速训练） #
#   training.play_steps=200 : 回放200步
#   training.play_render_mode=auto : 自动渲染模式
#   training.logger=tensorboard : 使用TensorBoard记录日志
#   training.device=null  : 自动选择gpu/cpu训练
# ========================================================

import mujoco
import mujoco.viewer

# 从 XML 文件加载（会读取最新的修改）
model = mujoco.MjModel.from_xml_path('src/unilab/assets/robots/cod_2026_balance/cod_2026_balance.xml')
data = mujoco.MjData(model)
print('🚀 MuJoCo 查看器启动')
print(f'✅ 加载模型: {model.ngeom} 个几何体, {model.njnt} 个关节')
print()
print('=== 主关节限位（确认是 ±2 rad） ===')
for i in range(model.njnt):
    jnt_name = model.joint(i).name
    if 'child' not in jnt_name and 'Wheel' not in jnt_name:
        jnt_range = model.jnt_range[i]
        print(f'  {jnt_name:25s} [{jnt_range[0]:.2f}, {jnt_range[1]:.2f}] rad')

print()
print('💡 操作:')
print('   - 左键拖动身体部位')
print('   - 或用右侧控制面板的滑块直接设置关节角度')
print('   - 空格键暂停/播放')

with mujoco.viewer.launch(model, data) as viewer:
    while viewer.is_running():
        mujoco.mj_step(model, data)
