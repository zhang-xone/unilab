# ========================================================
# 参数详解：
#   --algo ppo           : 使用PPO算法 #
#   --task go2_joystick_flat : 任务为Go2摇杆控制 #
#   --sim mujoco         : 使用MuJoCo后端 #
#   algo.max_iterations=200 : 训练200轮 #
#   algo.num_steps_per_env=24 : 每环境每轮采集24步
#   algo.num_envs=4096   : 4096个并行环境
#   algo.save_interval=50 : 每50轮保存一次checkpoint
#   algo.load_run="-1"   : 加载最新训练结果（继续训练）
#   algo.checkpoint=-1   : 使用最新检查点
#   training.no_play=true : 关闭回放（加速训练） #
#   training.play_steps=200 : 回放200步
#   training.play_render_mode=auto : 自动渲染模式
#   training.logger=tensorboard : 使用TensorBoard记录日志
#   training.device=null  : 自动选择gpu/cpu训练
# ========================================================

# 测试模型加载
uv run python << 'EOF'
import mujoco
import mujoco.viewer

model = mujoco.MjModel.from_xml_path('src/unilab/assets/robots/cod_2026_balance/scene_flat.xml')
data = mujoco.MjData(model)

print('=== 模型信息 ===')
print(f'  nq (自由度): {model.nq}')
print(f'  nu (执行器): {model.nu}')
print(f'  ngeom (碰撞体): {model.ngeom}')
print(f'  qpos 长度: {len(data.qpos)}')
print()

# 检查 home keyframe
print('=== home keyframe ===')
for i in range(model.nkey):
    name = model.key(i).name
    print(f'  [{i}] {name}: qpos = {model.key_qpos[i]}')

print()
print('=== 应用 home keyframe ===')
data.qpos = model.key_qpos[0]  # 应用 home
print(f'  base 位置: x={data.qpos[0]:.2f}, y={data.qpos[1]:.2f}, z={data.qpos[2]:.2f}')
print(f'  前 3 个关节角度: {data.qpos[7]:.2f}, {data.qpos[8]:.2f}, {data.qpos[9]:.2f}')

mujoco.mj_forward(model, data)

print()
print('=== 碰撞检查 ===')
print(f'  接触点数量: {data.ncon}')
for k in range(data.ncon):
    contact = data.contact[k]
    name1 = model.geom(contact.geom1).name
    name2 = model.geom(contact.geom2).name
    print(f'  接触 {k}: {name1} ↔ {name2}')

print()
print('=== 关节列表 ===')
for i in range(model.njnt):
    if i == 0:
        continue  # 跳过 free joint
    name = model.joint(i).name
    q_idx = model.jnt_qposadr[i]
    print(f'  [{i}] {name:30s} qpos[{q_idx}] = {data.qpos[q_idx]:.4f} rad')

print()
print('🚀 启动查看器！')
print('💡 按 Tab 打开控制面板，拖动 ctrl 滑块可以控制电机')

with mujoco.viewer.launch(model, data) as viewer:
    while viewer.is_running:
        mujoco.mj_step(model, data)
EOF
